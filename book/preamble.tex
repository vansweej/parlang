% Preamble for the ParLang book and the glyph test. LuaLaTeX only.
% All fonts use the HarfBuzz renderer so default-ignorable code points
% (U+FE0F, U+200D) are hidden by shaping instead of reported missing.
% Fallback chain: DejaVu -> DejaVu Sans -> Noto Sans CJK SC -> Noto Emoji.
% The flake (OSFONTDIR) guarantees all of these fonts.
\usepackage{fontspec}
\directlua{luaotfload.add_fallback("bookfallback", {"DejaVu Sans:mode=harf;", "Noto Sans CJK SC:mode=harf;", "Noto Emoji:mode=harf;"})}
\setmainfont{DejaVu Serif}[Renderer=HarfBuzz,RawFeature={fallback=bookfallback}]
\setsansfont{DejaVu Sans}[Renderer=HarfBuzz,RawFeature={fallback=bookfallback}]
\setmonofont{DejaVu Sans Mono}[Renderer=HarfBuzz,Scale=MatchLowercase,RawFeature={fallback=bookfallback}]

% A character missing from every font is an error, so -halt-on-error stops.
