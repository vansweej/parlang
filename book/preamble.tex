% Preamble for the ParLang book and the glyph test. LuaLaTeX only.
% All fonts use the HarfBuzz renderer so default-ignorable code points
% (U+FE0F, U+200D) are hidden by shaping instead of reported missing.
% Fallback chain: DejaVu -> DejaVu Sans -> Noto Sans CJK SC -> Noto Emoji.
% The flake (OSFONTDIR) guarantees all of these fonts.
% amssymb: pandoc writes GFM task-list items as \item[$\square$] / $\boxtimes$.
\usepackage{amsmath,amssymb}
\usepackage{fontspec}
\directlua{luaotfload.add_fallback("bookfallback", {"DejaVu Sans:mode=harf;", "Noto Sans CJK SC:mode=harf;", "Noto Emoji:mode=harf;"})}
\setmainfont{DejaVu Serif}[Renderer=HarfBuzz,RawFeature={fallback=bookfallback}]
\setsansfont{DejaVu Sans}[Renderer=HarfBuzz,RawFeature={fallback=bookfallback}]
\setmonofont{DejaVu Sans Mono}[Renderer=HarfBuzz,Scale=MatchLowercase,RawFeature={fallback=bookfallback}]

% A character missing from every font is an error, so -halt-on-error stops.
\tracinglostchars=3

\usepackage[a4paper,margin=25mm]{geometry}
\usepackage{xcolor}
\usepackage{graphicx}

\usepackage{array}

% Third-pass stretch for paragraphs with long unbreakable code tokens.
\setlength{\emergencystretch}{6em}
\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt plus 2pt minus 1pt}

% Table of contents: room for 3-digit page numbers and numbers like 10.10
% (book.cls defaults: \@pnumwidth 1.55em, \@tocrmarg 2.55em,
% section {1.5em}{2.3em}, subsection {3.8em}{3.2em}).
\makeatletter
\renewcommand*\@pnumwidth{2.2em}
\renewcommand*\@tocrmarg{3.2em}
\renewcommand*\l@section{\@dottedtocline{1}{1.5em}{3.0em}}
\renewcommand*\l@subsection{\@dottedtocline{2}{4.5em}{3.9em}}

% Ragged-right section headings: monospace code in a heading has spaces that
% cannot stretch, so justified headings overflow. Same values as book.cls
% (written without spaces after \@plus/\@minus, which chktex flags), plus
% \raggedright (chapter heads are already ragged-right).
\renewcommand\section{\@startsection{section}{1}{\z@}{-3.5ex\@plus-1ex\@minus-.2ex}{2.3ex\@plus.2ex}{\normalfont\Large\bfseries\raggedright}}
\renewcommand\subsection{\@startsection{subsection}{2}{\z@}{-3.25ex\@plus-1ex\@minus-.2ex}{1.5ex\@plus.2ex}{\normalfont\large\bfseries\raggedright}}
\renewcommand\subsubsection{\@startsection{subsubsection}{3}{\z@}{-3.25ex\@plus-1ex\@minus-.2ex}{1.5ex\@plus.2ex}{\normalfont\normalsize\bfseries\raggedright}}
\makeatother
